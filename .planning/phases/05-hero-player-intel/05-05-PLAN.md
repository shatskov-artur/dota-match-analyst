Create client/src/components/IntelTooltip.tsx.

The component:
- Receives `playerIntel: PlayerIntel` and `anchorRef: React.RefObject<HTMLElement>` props
- Uses `useLayoutEffect` (NOT useEffect) to measure anchor position and determine flip direction
- Default: above portrait (`bottom: calc(100% + 8px), left: 50%, transform: translateX(-50%)`)
- Flip: if anchorRef.current.getBoundingClientRect().top < 180 → below portrait (`top: calc(100% + 8px), left: 50%, transform: translateX(-50%)`)
- Is `position: absolute` — must be inside a `position: relative` parent WITHOUT overflow-hidden

```typescript
import { useLayoutEffect, useRef, useState } from 'react'
import { heroMapper } from '../utils/heroMapper'
// CRITICAL: import from '../utils/heroMapper' — NOT '@shared/heroMapper' (Vite bundler compat)
import type { PlayerIntel } from '../hooks/useMatchIntel'

interface IntelTooltipProps {
  playerIntel: PlayerIntel
  heroName: string        // localized hero name (from heroMapper on the played hero)
  anchorRef: React.RefObject<HTMLDivElement>
  isLoading?: boolean     // true while useMatchIntel data is in flight
}

export default function IntelTooltip({ playerIntel, heroName, anchorRef, isLoading = false }: IntelTooltipProps) {
  const [positionAbove, setPositionAbove] = useState(true)

  // CRITICAL: useLayoutEffect fires synchronously after DOM updates, before browser paint.
  // This prevents the single-frame position flash that useEffect would cause (Pitfall 3).
  useLayoutEffect(() => {
    if (!anchorRef.current) return
    const rect = anchorRef.current.getBoundingClientRect()
    // D-07 threshold: if portrait top < 180px from viewport top, flip tooltip below
    setPositionAbove(rect.top >= 180)
  }, [anchorRef])

  const positionStyle: React.CSSProperties = positionAbove
    ? { bottom: 'calc(100% + 8px)', left: '50%', transform: 'translateX(-50%)' }
    : { top: 'calc(100% + 8px)', left: '50%', transform: 'translateX(-50%)' }

  // Stat line — handles hidden profile (null) and loading states
  const statLine = () => {
    if (isLoading) {
      return <span style={{ fontSize: 10, color: '#444444' }}>Loading...</span>
    }
    if (playerIntel.games === null || playerIntel.winRate === null) {
      // PLAYER-02: hidden profile — show em dashes (U+2014)
      return (
        <span style={{ fontSize: 10, color: '#888888' }}>
          {'—'} games {'·'} {'—'}% on {heroName}
        </span>
      )
    }
    return (
      <span style={{ fontSize: 10, color: '#888888' }}>
        {playerIntel.games} games {'·'} {Math.round(playerIntel.winRate * 100)}% on {heroName}
      </span>
    )
  }

  const hasCounters = playerIntel.counters.length > 0

  return (
    <div
      style={{
        position: 'absolute',
        zIndex: 50,
        minWidth: 160,
        maxWidth: 220,
        background: '#111111',
        border: '1px solid #1a1a1a',
        borderRadius: 4,
        padding: 8,
        boxShadow: '0 4px 16px rgba(0,0,0,0.6)',
        pointerEvents: 'none',  // tooltip is read-only — no click capture
        ...positionStyle,
      }}
    >
      {/* Player name — top row, bold */}
      <p
        style={{
          fontSize: 10,
          fontWeight: 700,
          color: '#d8d8d8',
          margin: 0,
          marginBottom: 2,
          overflow: 'hidden',
          textOverflow: 'ellipsis',
          whiteSpace: 'nowrap',
        }}
      >
        {playerIntel.playerName}
      </p>

      {/* Stat line */}
      <p style={{ margin: 0, marginBottom: hasCounters ? 6 : 0 }}>
        {statLine()}
      </p>

      {/* Counters section — hidden when no counters */}
      {isLoading && (
        <>
          {/* Separator */}
          <div style={{ height: 1, background: '#1a1a1a', marginBottom: 6 }} />
          {/* Section label */}
          <p style={{ fontSize: 10, fontWeight: 700, color: '#444444', margin: 0, marginBottom: 4, textTransform: 'uppercase' }}>
            Counters
          </p>
          {/* Loading placeholders — no animate-pulse (05-UI-SPEC §Loading States) */}
          {[0, 1, 2].map(i => (
            <div key={i} style={{ display: 'flex', alignItems: 'center', gap: 4, marginBottom: 3 }}>
              <div style={{ width: 32, height: 32, background: '#1a1a1a', borderRadius: 2, flexShrink: 0 }} />
              <div style={{ height: 10, background: '#1a1a1a', borderRadius: 2, flex: 1 }} />
            </div>
          ))}
        </>
      )}

      {!isLoading && hasCounters && (
        <>
          {/* Separator */}
          <div style={{ height: 1, background: '#1a1a1a', marginBottom: 6 }} />
          {/* Section label */}
          <p style={{ fontSize: 10, fontWeight: 700, color: '#444444', margin: 0, marginBottom: 4, textTransform: 'uppercase' }}>
            Counters
          </p>
          {/* Top-3 counter rows (D-05) */}
          {playerIntel.counters.map((counter) => {
            const counterInfo = heroMapper(counter.heroId)
            const hasKnownPlayer = counter.knownPlayers.length > 0
            return (
              <div
                key={counter.heroId}
                style={{ display: 'flex', alignItems: 'center', gap: 4, marginBottom: 3 }}
              >
                {/* 32px mini-portrait */}
                {counterInfo ? (
                  <img
                    src={counterInfo.portrait}
                    alt={counterInfo.name}
                    width={32}
                    height={32}
                    style={{ width: 32, height: 32, objectFit: 'cover', borderRadius: 2, flexShrink: 0 }}
                  />
                ) : (
                  <div style={{ width: 32, height: 32, background: '#1a1a1a', borderRadius: 2, flexShrink: 0 }} />
                )}
                {/* Hero name + optional ⚠ flag */}
                <span style={{ fontSize: 10, color: '#d8d8d8' }}>
                  {counterInfo?.name ?? `Hero ${counter.heroId}`}
                </span>
                {hasKnownPlayer && (
                  <>
                    {/* ⚠ U+26A0 in dire red — not a team indicator, a warning */}
                    <span style={{ fontSize: 10, color: '#ef4444' }}>{'⚠'}</span>
                    {/* Opposing player name — truncated to 12 chars */}
                    <span
                      style={{
                        fontSize: 10,
                        color: '#888888',
                        overflow: 'hidden',
                        textOverflow: 'ellipsis',
                        whiteSpace: 'nowrap',
                        maxWidth: 80,
                      }}
                    >
                      {counter.knownPlayers[0].length > 12
                        ? counter.knownPlayers[0].slice(0, 12) + '…'
                        : counter.knownPlayers[0]}
                    </span>
                  </>
                )}
              </div>
            )
          })}
        </>
      )}
    </div>
  )
}
```

IMPORTANT NOTE on pickrate display:
The BFF returns `pick_rate` as raw `pro_pick` count (not a 0-1 percentage). The badge format
per 05-CONTEXT.md D-01 is `{winRate}% · {pickRate}%`. Since we don't have total pro games,
display raw pick count divided by a reasonable normalization. Per 05-UI-SPEC §Badge:
the pickrate value is always shown in `#888888` with no threshold coloring.
For v1: display the raw pick count directly as a number next to a pick icon, OR normalize
by assuming the max pick count as denominator. The simplest correct implementation per the
spec's format `52% · 18%` requires a percentage — but we only have raw count.
RESOLUTION: Show `pick_rate` directly as a raw number with a `P:` prefix (e.g. `52% · P:142`)
since we cannot compute a true percentage without total pro games. This is Claude's discretion
per 05-CONTEXT.md §Claude's Discretion. If the BFF is updated later to return a normalized
pick_rate (0-1), update the display accordingly.
SIMPLER ALTERNATIVE (recommended): Display format `{winRate}%` only for the winrate badge
(omit pickrate raw count since it's not meaningful to the user without context).
The spec says `52% · 18%` but that assumes a normalized pickrate. If pickrate is a raw count
(e.g. 142), show `52% · 142P` or just `52%`.

The executor should check what format the BFF actually returns (`pro_pick` raw count vs 0-1
normalized) and implement accordingly. If the BFF returns raw count and no normalization is
possible, show `{winRate}%` only (omit pickrate) rather than showing a misleading percentage.
Badge content decision is Claude's discretion (05-CONTEXT.md §Claude's Discretion).

```typescript
import { useRef, useState } from 'react'
import { heroMapper } from '../utils/heroMapper'
import { winrateColor } from '../utils/winrateColor'
import IntelTooltip from './IntelTooltip'
import type { PlayerIntel } from '../hooks/useMatchIntel'
import type { HeroStatsEntry } from '../hooks/useHeroStats'

interface DraftPortraitProps {
  kind: 'pick' | 'ban'
  heroId?: number
  isActive?: boolean
  ordinal?: string
  // Phase 5 additions:
  heroStats?: HeroStatsEntry            // from useHeroStats() — undefined while loading or on error
  playerIntel?: PlayerIntel             // from useMatchIntel() — undefined while loading or on error
}

export default function DraftPortrait({
  kind, heroId, isActive = false, ordinal, heroStats, playerIntel,
}: DraftPortraitProps) {
  const heroInfo = heroId !== undefined ? heroMapper(heroId) : null
  const anchorRef = useRef<HTMLDivElement>(null)
  const [showTooltip, setShowTooltip] = useState(false)

  const canShowTooltip = kind === 'pick' && heroId !== undefined && !!heroInfo && !!playerIntel

  if (!heroInfo) {
    // Empty placeholder — unchanged from Phase 4
    return (
      <div
        className={`w-14 h-14 shrink-0 rounded-sm${isActive ? ' animate-pulse' : ''}`}
        style={{
          background: '#141414',
          border: isActive ? '1px solid #b03030' : '1px solid #1e1e1e',
        }}
      />
    )
  }

  // Filled slot — outer wrapper is relative WITHOUT overflow-hidden (Pitfall 4 fix)
  return (
    <div
      ref={anchorRef}
      className="relative w-14 h-14 shrink-0"
      onMouseEnter={() => canShowTooltip && setShowTooltip(true)}
      onMouseLeave={() => setShowTooltip(false)}
      style={{ cursor: canShowTooltip ? 'default' : undefined }}

      {/* Inner portrait div — overflow-hidden for image + overlays */}

            aria-hidden="true"
```
