```typescript
function formatMmSs(seconds: number): string {
  const safe = Math.max(0, Math.floor(seconds))
  const m = Math.floor(safe / 60)
  const s = safe % 60
  return `${m}:${s.toString().padStart(2, '0')}`
}
function formatK(value: number): string {
  // 12345 → "12.3"
  return (value / 1000).toFixed(1)
}
```

```typescript
function SkeletonHistoryBlock({ gameDuration }: { gameDuration: number | undefined }) {
  const [, force] = useState(0)
  useEffect(() => {
    const id = setInterval(() => force(t => t + 1), 1000)
    return () => clearInterval(id)
  }, [])
  // Elapsed within the current 30s window — clamps to 0..30
  const elapsed = Math.min(30, Math.max(0, Math.floor((gameDuration ?? 0) % 30)))
  return (
    <section style={{ background: PANEL_BG, borderRadius: 6, padding: 12, height: 380, display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
      <span style={{ color: SECONDARY_FG, fontSize: 12 }}>
        Накапливаем историю… ({elapsed}/30с)
      </span>
    </section>
  )
}
```
